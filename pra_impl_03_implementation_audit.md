## 7. Lane Count Formula Definitions

```javascript
// Computed from data.recommendations array

function computeLaneCounts(recs) {
  let action = 0, blocked = 0, anchor = 0, narrative = 0, explainability = 0, observation = 0;

  const CONVICTION_TYPES = new Set([
    "STRATEGIC_RETAIN_SIGNAL",
    "STRATEGIC_RETAIN_NARRATIVE",
    "CONVICTION_EXPLAINABILITY_CARD",
  ]);
  const NARRATIVE_TYPES = new Set([
    "PORTFOLIO_CONSTRUCTION_NARRATIVE",
    "THEMATIC_SATURATION_NARRATIVE",
  ]);
  const EXPLAINABILITY_TYPES = new Set([
    "REPLAY_ALIGNMENT_CONTEXT",
  ]);

  for (const r of recs) {
    const ct = r.card_type || "DIAGNOSTIC";
    const es = r.execution_state || "EXECUTABLE";
    const rt = r.recommendation_type || "";

    if (CONVICTION_TYPES.has(rt))          { anchor++; }
    else if (NARRATIVE_TYPES.has(rt))      { narrative++; }
    else if (EXPLAINABILITY_TYPES.has(rt)) { explainability++; }
    else if (ct === "ACTION") {
      if (es === "BLOCKED_BY_POLICY" || es === "DEFERRED_BY_POLICY") { blocked++; }
      else { action++; }
    }
    else if (ct === "OBSERVATION")         { observation++; }
    else                                   { observation++; }  // safe default
  }

  return { action, blocked, anchor, narrative, explainability, observation };
}
```
